const params = new URLSearchParams(location.search); function shuffle(arr) { for (let i = arr.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [arr[i], arr[j]] = [arr[j], arr[i]]; } return arr; } // ✅ читаем ВСЮ базу const pairsBase = JSON.parse(params.get('pairs') || '[]'); const limit = parseInt(params.get('limit') || pairsBase.length, 10); const extraCount = parseInt(params.get('extraCount') || '0', 10); // ✅ на каждый запуск страницы — новая выборка shuffle(pairsBase); // ✅ пары в игре const activePairs = pairsBase.slice(0, Math.min(limit, pairsBase.length)); const rejectedPairs = pairsBase.slice(activePairs.length); // ✅ делаем data в том формате, который у тебя уже был (A -> B) const dataObj = {}; activePairs.forEach(p => { dataObj[p.a] = p.b; }); // ✅ делаем extra (ловушки) как раньше: одиночки из оставшихся пар const pool = []; rejectedPairs.forEach(p => { pool.push(Math.random() > 0.5 ? p.a : p.b); }); shuffle(pool); const extraArr = pool.slice(0, Math.max(0, extraCount));
